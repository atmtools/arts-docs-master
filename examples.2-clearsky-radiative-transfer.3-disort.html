37ws.spectral_rad_transform_operatorSet(option="Tb")
38ws.spectral_rad_space_agendaSet(option="UniformCosmicBackground")
39ws.spectral_rad_surface_agendaSet(option="Blackbody")

42ws.disort_settings_agendaSetup()

49    disort_fourier_mode_dimension=1,
50)

53ws.ray_pathGeometricDownlooking(

55    lon=lon,
56    max_stepsize=1000.0,
57)
58ws.spectral_radClearskyEmission()
59
60# %% Plot results
61f = ws.disort_spectral_rad_field.freq_grid
62
63fig, ax = pyarts.plot(ws.disort_spectral_rad_field,
64                      plotstyle='plot', select='down', alpha=0.5)
65ax.semilogy(f, ws.spectral_rad[:, 0], "k--", lw=3)
66ax.semilogy(
67    f,
68    ws.disort_spectral_rad_field.data[:, 0, 0, (NQuad // 2) - 1],
69    "g:",
70    lw=3,
71)
72ax.semilogy(
73    f,
74    ws.disort_spectral_rad_field.data[:, 0, 0, 0],
75    "m:",
76    lw=3,
77)
78ax.set_xticks(f[::20], (f[::20] / 1e9).round(1))
79ax.set_ylabel("Spectral radiance [W sr$^{-1}$ m$^{-2}$ Hz$^{-1}$]")
80ax.set_xlabel("Dirac frequency [GHz]")
81ax.set_title("Downlooking")

  3import matplotlib.pyplot as plt
  4import numpy as np
  5import pyarts3 as pyarts
  6
  7if pyarts.arts.globals.data.is_lgpl:

 13# Download catalogs
 14pyarts.data.download()
 15
 16NQuad = 16
 17max_level_step = 1e3
 18atm_latitude = 0.0
 19atm_longitude = 0.0
 20solar_latitude = 0.0
 21solar_longitude = 0.0
 22surf_temperature = 293.0
 23surf_reflectivity = 0.05
 24cutoff = ["ByLine", 750e9]
 25remove_lines_percentile = 70
 26sunfile = "star/Sun/solar_spectrum_QUIET.xml"
 27planet = "Earth"
 28
 29# Input file names are prefixed with the example number
 30prefix = "3-"
 31
 32xarr = pyarts.data.xarray_open_dataset(prefix + "atmosphere.nc")
 33
 34ws = pyarts.Workspace()
 35
 36ws.freq_grid = pyarts.arts.convert.kaycm2freq(np.linspace(500, 2500, 1001))
 37ws.atm_field = pyarts.data.to_atm_field(xarr)
 38
 39v = pyarts.data.to_abs_species(ws.atm_field)
 40
 41ws.abs_species = v
 42ws.ReadCatalogData(ignore_missing=True)
 43ws.spectral_propmat_agendaAuto(T_extrapolfac=1e9)
 44
 45for band in ws.abs_bands:
 46    ws.abs_bands[band].cutoff = cutoff[0]
 47    ws.abs_bands[band].cutoff_value = cutoff[1]
 48
 49ws.abs_bands.keep_hitran_s(remove_lines_percentile)
 50
 51ws.surf_fieldPlanet(option=planet)
 52
 53sun = pyarts.arts.GriddedField2.fromxml(sunfile)
 54
 55ws.surf_field["t"] = surf_temperature
 56
 57ws.sunFromGrid(
 58    sun_spectrum_raw=sun,
 59    lat=solar_latitude,
 60    lon=solar_longitude,
 61)
 62
 63ws.disort_quadrature_dimension = NQuad
 64ws.disort_fourier_mode_dimension = 1
 65ws.disort_legendre_polynomial_dimension = 1
 66
 67ws.ray_pathGeometricDownlooking(
 68    lat=atm_latitude,
 69    lon=atm_longitude,
 70    max_stepsize=max_level_step,
 71)
 72
 73ws.disort_settings_agendaSetup(
 74    sun_setting="Sun",
 75    surf_setting="ThermalLambertian",
 76    surf_lambertian_value=surf_reflectivity * np.ones_like(ws.freq_grid),
 77)
 78
 79ws.disort_spectral_flux_fieldFromAgenda()
