Lines

 1"""
 2This file will showcase how you can load line data from arts-cat-data into the
 3workspace and do the required setups to perform a simple forward calculations
 4using this data.
 5"""
 6
 7import os
 8
 9import matplotlib.pyplot as plt
10import numpy as np
11import pyarts3 as pyarts
12
13# Initialize ARTS workspace
14ws = pyarts.workspace.Workspace()
15
16"""
17The workspace contains an absorption species helper list
18that allows it to understand what data and methods you
19will want to call.
20
21This example sets the absorption species to all oxygen
22and all water isotopologues (in their line-by-line mode).
23"""
24ws.abs_speciesSet(species=["O2-66", "H2O-161"])
25
26"""
27We now need to load the data.
28
29ARTS comes with a lot of line-by-line data available
30in its arts-cat-data repository.  The first line below
31downloads that data to a local cache and allows you
32to run this script.
33
34You can also provide line data in other ways, e.g., by loading HITRAN
35or some other format.  What is important is to populate the absorption
36bands with appropriate data.
37"""
38pyarts.data.download()
39ws.abs_bandsReadSpeciesSplitCatalog(basename="lines/")
40
41"""
42Compute absorption
43
44Now we can use any of the methods available to compute line-by-line absorption.
45
46Below we simply set a simple atmosphere and plot the resulting absorption.
47
48Please see other examples for more details on how to use line-by-line
49calculations in increasingly complex ways to solve your problem
50"""
51
52# Initialize an atmospheric point
53atm = pyarts.arts.AtmPoint()
54atm.temperature = 295  # At room temperature
55atm.pressure = 1e5  # At 1 bar
56atm["O2"] = 0.21  # At 21% atmospheric Oxygen
57atm["H2O"] = 0.001  # At 0.1% atmospheric Water Vapor
58
59# Set a frequency range and remove lines outside (to speed up calculations)
60freqs = np.linspace(1e9, 1000e9, 1001)
61ws.abs_bands.keep_frequencies(fmax=freqs[-1])
62
63# Use available plotting routines to draw the results
64fig, ax = pyarts.plot(
65    ws.abs_bands,
66    freqs=freqs,
67    atm=atm,
68    mode='important fill isotopes',
69    min_pm=1e-7)
70ax.legend(bbox_to_anchor=(1.05, 1))
71ax.set_xlim(freqs[0], freqs[-1])
72ax.set_ylim(ymin=1e-7, ymax=1)
73ax.set_xticks(np.linspace(0, 1000e9, 11), np.linspace(0, 1000, 11))
74ax.set_yscale('log')
75ax.set_xlabel("Frequency [GHz]")
76ax.set_ylabel("Absorption [1/m]")
77ax.set_title("Absorption by species")
78
79if "ARTS_HEADLESS" not in os.environ:
80    plt.show()

Cia

 1"""
 2
 3This file will showcase how you can load collision-induced absorption (CIA)
 4data from arts-cat-data into the workspace and do the required setups to
 5perform a simple forward calculations using this data
 6
 7Note that this example presumes that you have set the environment variable
 8ARTS_DATA_PATH to contain a path to a local copy of both arts-cat-data and
 9arts-xml-data before you import pyarts3.  Please check that this is the case
10if the example does not work for you.  You can easily check if this path is
11set by adding the following two lines at the top of this pyarts-controlfile:
12
13```
14import os
15print(os.environ.get("ARTS_DATA_PATH"))
16```
17
18"""
19
20import os
21
22import matplotlib.pyplot as plt
23import numpy as np
24import pyarts3 as pyarts
25
26# Initialize ARTS workspace
27ws = pyarts.workspace.Workspace()
28
29"""
30The workspace contains an absorption species helper list
31that allows it to understand what data and methods you
32will want to call.
33
34This example sets the absorption species to O2 collision-induced
35absorption (CIA) with self and also with N2
36"""
37ws.abs_speciesSet(species=["O2-CIA-O2", "O2-CIA-N2"])
38
39"""
40We now need to load the data.
41
42ARTS comes with a lot of CIA data available
43in its arts-cat-data repository.  The first line below
44downloads that data to a local cache and allows you
45to run this script.
46
47You can also provide CIA data in other ways, e.g., by loading HITRAN
48or some other format.  What is important is to populate the absorption
49CIA data object with appropriate data.
50"""
51pyarts.data.download()
52ws.abs_cia_dataReadSpeciesSplitCatalog(basename="cia/")
53
54"""
55Compute absorption
56
57Now we can use any of the methods available to compute CIA absorption.
58
59Below we simply set a simple atmosphere and plot the resulting absorption.
60
61Please see other examples for more details on how to use CIA
62calculations in increasingly complex ways to solve your problem
63"""
64
65# Initialize an atmospheric point
66atm = pyarts.arts.AtmPoint()
67atm.temperature = 295  # At room temperature
68atm.pressure = 1e5  # At 1 bar
69atm["O2"] = 0.21  # At 21% atmospheric Oxygen
70atm["N2"] = 0.78  # At 78% atmospheric Nitrogen
71
72# Plot the absorption of this example
73fig, ax = pyarts.plot(ws.abs_cia_data, atm=atm)
74for i, a in enumerate(ax.flatten()):
75    a.set_xlabel("Frequency [Hz]")
76    a.set_ylabel("Absorption [1/m]")
77    a.set_title(a.get_lines()[0].get_label())
78fig.suptitle("CIA Absorption")
79
80if "ARTS_HEADLESS" not in os.environ:
81    plt.tight_layout()
82    plt.show()

Single species recipe

 1"""Absorption by a single species"""
 2
 3import os
 4
 5import matplotlib.pyplot as plt
 6import numpy as np
 7import pyarts3 as pyarts
 8
 9# Download catalogs
10pyarts.data.download()
11
12# %% Select absorption species
13species = "O2-66"  # Main isotope of O2
14
15# %% Activate the recipe for this species
16#
17# See [SingleSpeciesAbsorption](pyarts3.recipe.rst#pyarts3.recipe.SingleSpeciesAbsorption).
18absorption = pyarts.recipe.SingleSpeciesAbsorption(species=species)
19
20# %% Select a single temperature, a VMR value, and a range of pressures
21atm = pyarts.arts.AtmPoint()
22atm["O2"] = 0.2095
23atm.temperature = 273
24ps = np.logspace(5, -2, 8)
25
26# %% Select frequency range
27line_f0 = 118750348044.712  # Lowest energy absorption line
28f = np.linspace(-500e6, 500e6, 1001) + line_f0  # Some range around it
29
30# %% Use the recipe and convert the results to cross-sections
31xsec = []
32for p in ps:
33    atm.pressure = p
34    xsec.append(absorption(f, atm) / atm.number_density(species))
35xsec = np.array(xsec)
36
37# %% Plot the results
38fig, ax = plt.subplots()
39ax.semilogy((f - line_f0) / 1e6, xsec.T)
40ax.set_xlabel("Frequency offset [MHz]")
41ax.set_ylabel("Cross-section [$m^2$]")
42ax.set_title("Cross-section of O$_2$ 16-16")
43ax.set_ylim(ymin=1e-3 * np.min(xsec))
44ax.legend(
45    [f"P: $10^{'{'}{round(np.log10(x))}{'}'}$" for x in ps],
46    ncols=4,
47    loc="lower center",
48)
49
50if "ARTS_HEADLESS" not in os.environ:
51    plt.show()
52
53# %% Integration test by ensuring some statistics look good
54assert np.isclose(6.792977548868407e-28 / xsec.mean(), 1)
55assert np.isclose(5.43981642113382e-24 / xsec.sum(), 1)
56assert np.isclose(1.3359834491781882e-24 / xsec.max(), 1)
57assert np.isclose(2.537911691540087e-26 / xsec.std(), 1)
58assert np.isclose(8.236637542411964e-35 / xsec.min(), 1)
