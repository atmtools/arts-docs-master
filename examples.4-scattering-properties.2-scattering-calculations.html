Scattering calculations

  1"""
  2Scattering calculations
  3
  4This example demonstrates the basic principles of performing scattering
  5calcualtions with ARTS. Populations of particles that scatter radiation are
  6referred to as *scattering species*. Each scattering species defines a mapping
  7from one or several fields of *scattering species properties* to *scattering
  8properties* which form the input to the actual scattering calculation.
  9"""
 10
 11import os
 12
 13import matplotlib.pyplot as plt
 14import numpy as np
 15import pyarts3 as pyarts
 16from pyarts3.arts import ArrayOfScatteringSpecies, HenyeyGreensteinScatterer
 17
 18# Download catalogs
 19pyarts.data.download()
 20
 21# %% Set up a 1D atmosphere
 22ws = pyarts.Workspace()
 23ws.surf_fieldPlanet(option="Earth")
 24ws.surf_field[pyarts.arts.SurfaceKey("t")] = 295.0
 25ws.atm_fieldRead(
 26    toa=100e3, basename="planets/Earth/afgl/tropical/", missing_is_zero=1
 27)
 28ws.atm_fieldIGRF(time="2000-03-11 14:39:37")
 29
 30# %% Add a field of scattering species properties
 31#
 32# For this example, we will use scatterers with a Henyey-Greenstein phase
 33# function to represent ice particles. We will use two *scattering species
 34# properties* to represent the scattering species *ice*: The extinction and
 35# the single-scattering albed (SSA). The ice extinction and SSA thus become
 36# atmospheric fields. Below we define scattering-species-property object that
 37# identify these fields.
 38ice_extinction = pyarts.arts.ScatteringSpeciesProperty(
 39    "ice", pyarts.arts.ParticulateProperty("Extinction")
 40)
 41ice_ssa = pyarts.arts.ScatteringSpeciesProperty(
 42    "ice", pyarts.arts.ParticulateProperty("SingleScatteringAlbedo")
 43)
 44
 45# We then define a GriddedField3 representing the ice extinction and
 46# single-scattering albedo and add it to ``atm_field`` of the workspace.
 47grids = ws.atm_field["t"].data.grids
 48z = grids[0]
 49ice_extinction_profile = np.zeros_like(z)
 50ice_extinction_profile[(z > 5e3) * (z < 15e3)] = 1.0
 51ice_extinction_profile = pyarts.arts.GriddedField3(
 52    data=ice_extinction_profile[..., None, None], grids=grids
 53)
 54ws.atm_field[ice_extinction] = ice_extinction_profile
 55
 56ice_ssa_profile = np.zeros_like(z)
 57ice_ssa_profile[(z > 5e3) * (z < 15e3)] = 0.5
 58ice_ssa_profile = pyarts.arts.GriddedField3(
 59    data=ice_ssa_profile[..., None, None], grids=grids
 60)
 61ws.atm_field[ice_ssa] = ice_ssa_profile
 62
 63# %% Rain
 64rain_extinction = pyarts.arts.ScatteringSpeciesProperty(
 65    "rain", pyarts.arts.ParticulateProperty("Extinction")
 66)
 67rain_ssa = pyarts.arts.ScatteringSpeciesProperty(
 68    "rain", pyarts.arts.ParticulateProperty("SingleScatteringAlbedo")
 69)
 70
 71rain_extinction_profile = np.zeros_like(z)
 72rain_extinction_profile[z < 5e3] = 1.0
 73rain_extinction_profile = pyarts.arts.GriddedField3(
 74    data=rain_extinction_profile[..., None, None], grids=grids
 75)
 76ws.atm_field[rain_extinction] = rain_extinction_profile
 77
 78rain_ssa_profile = np.zeros_like(z)
 79rain_ssa_profile[z < 5e3] = 0.5
 80rain_ssa_profile = pyarts.arts.GriddedField3(
 81    data=rain_ssa_profile[..., None, None], grids=grids
 82)
 83ws.atm_field[rain_ssa] = rain_ssa_profile
 84
 85# %% Create the scattering species
 86
 87hg_ice = HenyeyGreensteinScatterer(ice_extinction, ice_ssa, 0.5)
 88hg_rain = HenyeyGreensteinScatterer(rain_extinction, rain_ssa, 0.0)
 89scattering_species = ArrayOfScatteringSpecies()
 90scattering_species.add(hg_ice)
 91scattering_species.add(hg_rain)
 92
 93
 94# ## Extracting scattering properties
 95#
 96# We can now extract bulk scattering properties from the scattering species.
 97
 98# %% Ice
 99#
100# We can extract the ice phase function from the combind scattering species
101# by calculating the bulk-scattering properties at an altitude above 5 km. To
102# check the consistency of the Henyey-Greenstein scatterer, we extract the data
103# in gridded and spectral representation and ensure that they are the same when
104# both are converted to gridded representation.
105atm_pt = ws.atm_field(11e3, 0.0, 0.0)
106f_grid = np.array([89e9])
107bsp = scattering_species.get_bulk_scattering_properties_tro_spectral(atm_pt, f_grid, 32)
108pm_spectral = bsp.phase_matrix
109
110za_scat_grid = pyarts.arts.ScatteringTroSpectralVector.to_gridded(
111    pm_spectral
112).get_za_scat_grid()
113bsp = scattering_species.get_bulk_scattering_properties_tro_gridded(
114    atm_pt, f_grid, za_scat_grid
115)
116pm_gridded = bsp.phase_matrix
117
118# %% Plot
119fig, ax = plt.subplots()
120ax.plot(
121    pyarts.arts.ScatteringTroSpectralVector.to_gridded(pm_spectral).flatten()[::6],
122    label="Converted from Spectral",
123)
124ax.plot(pm_gridded.flatten()[::6], ls="--", label="Gridded")
125ax.set_title("Henyey-Greenstein phase function")
126ax.legend()
127
128# %% Rain
129#
130# Similarly, we can extract the phase function for rain by calculating the
131# bulk-scattering properties at a lower point in the atmosphere.
132atm_pt = ws.atm_field(4e3, 0.0, 0.0)
133bsp = scattering_species.get_bulk_scattering_properties_tro_spectral(atm_pt, f_grid, 32)
134pm_spectral = bsp.phase_matrix
135
136za_scat_grid = pyarts.arts.ScatteringTroSpectralVector.to_gridded(
137    pm_spectral
138).get_za_scat_grid()
139bsp = scattering_species.get_bulk_scattering_properties_tro_gridded(
140    atm_pt, f_grid, za_scat_grid
141)
142pm_gridded = bsp.phase_matrix
143
144# %% Plot
145fig, ax = plt.subplots()
146ax.plot(
147    pyarts.arts.ScatteringTroSpectralVector.to_gridded(pm_spectral).flatten()[::6],
148    label="Converted from Spectral",
149)
150ax.plot(pm_gridded.flatten()[::6], ls="--", label="Gridded")
151ax.set_title("Henyey-Greenstein phase function")
152ax.legend()
153
154if "ARTS_HEADLESS" not in os.environ:
155    plt.show()
